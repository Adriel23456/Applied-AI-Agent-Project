% ---------- Safe input / graphics / listings ----------
\newcommand{\missingfilebox}[1]{%
  \par\medskip\noindent
  \fcolorbox{red!60!black}{red!5}{%
    \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule}{%
      \centering\ttfamily\small MISSING FILE: \detokenize{#1}}}%
  \par\medskip}

\newcommand{\missingimagebox}[1]{%
  \begingroup
  \setlength{\fboxsep}{0pt}%
  \fcolorbox{tecrule}{boxbg}{%
    \parbox[c][2.8cm][c]{6cm}{\centering\ttfamily\scriptsize
      IMAGE NOT FOUND\\[3pt]\detokenize{#1}}}%
  \endgroup}

\newcommand{\safeinput}[1]{%
  \IfFileExists{#1.tex}{\input{#1}}{%
    \PackageWarning{tecdoc}{Missing section file #1.tex}%
    \missingfilebox{#1.tex}}}

\newcommand{\safeimage}[2][width=0.75\linewidth]{%
  \IfFileExists{#2}{\includegraphics[#1]{#2}}{%
    \IfFileExists{img/#2}{\includegraphics[#1]{img/#2}}{%
      \IfFileExists{figures/#2}{\includegraphics[#1]{figures/#2}}{%
        \PackageWarning{tecdoc}{Missing image #2}%
        \missingimagebox{#2}}}}}

\newcommand{\safelisting}[2][]{%
  \IfFileExists{#2}{\lstinputlisting[#1]{#2}}{%
    \PackageWarning{tecdoc}{Missing code file #2}%
    \missingfilebox{#2}}}
