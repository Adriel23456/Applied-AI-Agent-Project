\newcommand{\missingfilebox}[1]{%
  \par\medskip\noindent
  \fcolorbox{red!60!black}{red!5}{%
    \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule}{%
      \centering\ttfamily\scriptsize MISSING FILE: \detokenize{#1}}}%
  \par\medskip}

\newcommand{\missingimagebox}[1]{%
  \begingroup\setlength{\fboxsep}{0pt}%
  \fcolorbox{tecrule}{boxbg}{%
    \parbox[c][2.4cm][c]{0.9\linewidth}{\centering\ttfamily\scriptsize
      IMAGE NOT FOUND\\[3pt]\detokenize{#1}}}%
  \endgroup}

\newcommand{\safeinput}[1]{%
  \IfFileExists{#1.tex}{\input{#1}}{%
    \PackageWarning{tecpaper}{Missing section file #1.tex}%
    \missingfilebox{#1.tex}}}

\newcommand{\safeimage}[2][width=\columnwidth]{%
  \IfFileExists{#2}{\includegraphics[#1]{#2}}{%
    \IfFileExists{img/#2}{\includegraphics[#1]{img/#2}}{%
      \PackageWarning{tecpaper}{Missing image #2}%
      \missingimagebox{#2}}}}

\newcommand{\safelisting}[2][]{%
  \IfFileExists{#2}{\lstinputlisting[#1]{#2}}{%
    \PackageWarning{tecpaper}{Missing code file #2}%
    \missingfilebox{#2}}}
